    %% ==========================================
    %% --- CODRU ACTUALIZAT JAVASCRIPT / JPA ---

    class User {
        +Long id
        +String username
        +String email
        +String password
        +UserType userType
    }

    class Person {
        +Long id
        +String firstName
        +String lastName
        +LocalDate dateOfBirth
        +String gender
        +String country
        +String city
        +DocumentType documentType
        +String documentId
        +String phoneNumber
    }

    class AppOrder {
        +Long id
        +OrderType orderType
        +Double amountPaid
        +LocalDateTime purchaseDate
    }

    class Ticket {
        +Long id
        +TicketType ticketType
        +Double price
        +LocalDateTime purchaseDate
        +boolean checkInCompleted
    }

    class Bracelet {
        +Long id
        +String rfidTag
        +Double balance
    }

    %% ==========================================
    %% --- SCENE, ARTIȘTI & PLAYLIST MELODII ---
    %% ==========================================
    class Stage {
        +Long id
        +String name
        +Integer capacity
        +String locationDescription
        +Double latitude
        +Double longitude
        +Double radiusMeters
    }

    class Artist {
        +Long id
        +String name
        +String genre
        +String biography
        +String imageUrl
        +String spotifyId
    }

    class Song {
        +Long id
        +String title
        +Integer durationSeconds
        +String spotifyId
        +String previewUrl
    }

    class Performance {
        +Long id
        +LocalDateTime startTime
        +LocalDateTime endTime
        +PerformanceStatus status
    }

    class StageCrowdReport {
        +Long id

    }

    class SOSAlert {
        +Long id
        +LocalDateTime timestamp

        +Double longitude
        +SOSType type
        +Boolean resolved
    }

    %% ==========================================
    %% --- RELAȚII ÎNTRE ENTITĂȚI ---
    %% ==========================================
    User "1" --> "*" AppOrder : plaseaza comenzi
    AppOrder "*" --> "1" Ticket : contine biletul cumparat
    Ticket "1" --> "*" AppOrder : are comenzi asociate
    Ticket "1" --> "0..1" Person : are ca posesor (attendee)
    Ticket "1" <--> "0..1" Bracelet : este asociat la bratara (rfid/token)
    
    Stage "1" --> "*" Performance : gazduieste evenimente
    Artist "1" --> "*" Performance : canta in
    Performance "*" --> "1" Artist : are ca invitat
    Performance "*" --> "1" Stage : are loc pe
    Artist "1" --> "*" Song : are in playlistul de melodii (Audio preview)

    User "1" --> "*" UserFavorite : selecteaza favoriti
    UserFavorite "*" --> "1" Artist : indica artistul preferat

    User "1" --> "*" PersonalScheduleSlot : isi organizeaza calendarul (Clash Detector)
    PersonalScheduleSlot "*" --> "1" Performance : asociaza slotul de timp
    
    User "1" --> "*" StageCrowdReport : raporteaza aglomeratia
    StageCrowdReport "*" --> "1" Stage : indica scena monitorizata

    User "1" --> "*" SOSAlert : trimite semnale de urgenta GPS
